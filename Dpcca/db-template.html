// ============================================================
// CloudBase 文档数据库 CRUD 封装（Web SDK v2）
// 与 Dpcca/rating.html 的 dataProvider 完全同构
// ============================================================
const ENV_ID = "dpcca-wiki-d7g0dl19y23cd30f3";

let _app = null, _auth = null, _db = null;

/** 初始化数据库（幂等），返回 db 实例 */
async function getDB() {
  if (_db) return _db;
  if (typeof cloudbase === "undefined")
    throw new Error("请先引入 @cloudbase/js-sdk");
  _app = cloudbase.init({ env: ENV_ID });
  _auth = _app.auth({ persistence: "local" });
  if (!_auth.hasLoginState()) {
    await _auth.anonymousAuthProvider().signIn(); // 匿名登录
  }
  _db = _app.database();
  return _db;
}

/** 命令操作符（$inc / $push / $gt 等） */
async function cmd() {
  const db = await getDB();
  return db.command;
}

// ---------- 增 ----------
/** 新增文档，返回新文档 _id；docId 可选，指定则用该 ID（存在则覆盖） */
async function addDoc(collection, data, docId) {
  const db = await getDB();
  const res = docId
    ? await db.collection(collection).doc(docId).set(data)
    : await db.collection(collection).add({ data });
  return docId || res.id;
}

// ---------- 查 ----------
/** 查询全部（默认 20 条） */
async function listDocs(collection, limit = 20) {
  const db = await getDB();
  const res = await db.collection(collection).limit(limit).get();
  return res.data;
}

/** 按 _id 查询，返回文档或 null */
async function getDoc(collection, id) {
  const db = await getDB();
  const res = await db.collection(collection).doc(id).get();
  const item = Array.isArray(res.data) ? res.data[0] : res.data;
  return item || null;
}

/** 条件查询；where 传普通对象（支持 $gt/$in/$and 等，见下方示例） */
async function queryDocs(collection, where, { orderField, order = "asc", limit = 20 } = {}) {
  const db = await getDB();
  let q = db.collection(collection);
  if (where && Object.keys(where).length) q = q.where(where);
  if (orderField) q = q.orderBy(orderField, order);
  const res = await q.limit(limit).get();
  return res.data;
}

// ---------- 改 ----------
/** 按 _id 更新部分字段；data 支持命令操作符值 */
async function updateDoc(collection, id, data) {
  const db = await getDB();
  const res = await db.collection(collection).doc(id).update(data);
  return res.updated; // 更新的记录数
}

/** 条件批量更新 */
async function updateDocsWhere(collection, where, data) {
  const db = await getDB();
  const res = await db.collection(collection).where(where).update(data);
  return res.updated;
}

// ---------- 删 ----------
/** 按 _id 删除 */
async function removeDoc(collection, id) {
  const db = await getDB();
  await db.collection(collection).doc(id).remove();
  return true;
}

/** 条件批量删除 */
async function removeDocsWhere(collection, where) {
  const db = await getDB();
  await db.collection(collection).where(where).remove();
  return true;
}
